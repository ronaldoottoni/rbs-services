Tabelas Banco De Dados
CREATE TYPE "PessoaTipo" AS ENUM (
  'CLIENTE',
  'FORNECEDOR',
  'FUNCIONARIO',
  'TRANSPORTADORA',
  'BANCO'
);

CREATE TYPE "PropriedadeTipo" AS ENUM (
  'FAZENDA',
  'SITIO',
  'CHACARA',
  'OUTRO'
);

CREATE TABLE "Pessoa" (
  "id" TEXT PRIMARY KEY,
  "tipo" "PessoaTipo" NOT NULL,
  "cpfCnpj" TEXT,
  "nome" TEXT NOT NULL,
  "telefone" TEXT,
  "email" TEXT,
  "cidade" TEXT,
  "estado" TEXT,
  "ativo" BOOLEAN NOT NULL DEFAULT TRUE,
  "createdAt" TIMESTAMP NOT NULL DEFAULT NOW(),
  "updatedAt" TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE "Propriedade" (
  "id" TEXT PRIMARY KEY,
  "clienteId" TEXT NOT NULL,
  "nome" TEXT NOT NULL,
  "tipo" "PropriedadeTipo" NOT NULL,
  "matricula" TEXT,
  "areaTotalHa" DOUBLE PRECISION,
  "cidade" TEXT,
  "estado" TEXT,
  "endereco" TEXT,
  "latitude" DOUBLE PRECISION,
  "longitude" DOUBLE PRECISION,
  "ativo" BOOLEAN NOT NULL DEFAULT TRUE,
  "createdAt" TIMESTAMP NOT NULL DEFAULT NOW(),
  "updatedAt" TIMESTAMP NOT NULL DEFAULT NOW(),

  CONSTRAINT "Propriedade_clienteId_fkey"
    FOREIGN KEY ("clienteId")
    REFERENCES "Pessoa"("id")
);

CREATE TYPE "AreaTipo" AS ENUM (
  'CULTIVO',
  'PASTAGEM',
  'RESERVA',
  'APP',
  'SEDE',
  'ESTRADA',
  'AGUA',
  'OUTRO'
);

CREATE TABLE "AreaPropriedade" (
  "id" TEXT PRIMARY KEY,
  "propriedadeId" TEXT NOT NULL,
  "nome" TEXT NOT NULL,
  "tipo" "AreaTipo" NOT NULL,
  "areaHa" DOUBLE PRECISION,
  "culturaAtual" TEXT,
  "descricao" TEXT,
  "latitudeCentro" DOUBLE PRECISION,
  "longitudeCentro" DOUBLE PRECISION,
  "coordenadasGeoJson" JSONB,
  "ativo" BOOLEAN NOT NULL DEFAULT TRUE,
  "createdAt" TIMESTAMP NOT NULL DEFAULT NOW(),
  "updatedAt" TIMESTAMP NOT NULL DEFAULT NOW(),

  CONSTRAINT "AreaPropriedade_propriedadeId_fkey"
    FOREIGN KEY ("propriedadeId")
    REFERENCES "Propriedade"("id")
);

CREATE TYPE "ServicoTipo" AS ENUM (
  'DRONE',
  'AMBIENTAL',
  'IRRIGACAO',
  'CONSULTORIA',
  'OUTRO'
);

CREATE TYPE "ServicoUnidade" AS ENUM (
  'HECTARE',
  'HORA',
  'DIARIA',
  'PROJETO',
  'UNIDADE'
);

CREATE TABLE "ServicoCatalogo" (
  "id" TEXT PRIMARY KEY,
  "nome" TEXT NOT NULL,
  "tipo" "ServicoTipo" NOT NULL,
  "descricao" TEXT,
  "valorPadrao" DOUBLE PRECISION,
  "unidade" "ServicoUnidade" NOT NULL,
  "ativo" BOOLEAN NOT NULL DEFAULT TRUE,
  "createdAt" TIMESTAMP NOT NULL DEFAULT NOW(),
  "updatedAt" TIMESTAMP NOT NULL DEFAULT NOW()
);
